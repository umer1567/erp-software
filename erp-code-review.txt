import { NextResponse } from "next/server";
import { jwtVerify } from "jose";

const secret = new TextEncoder().encode(
  process.env.JWT_SECRET
);

export async function middleware(request) {
  const token =
    request.cookies.get(
      "erp_token"
    )?.value;

  if (!token) {
    return NextResponse.redirect(
      new URL(
        "/login",
        request.url
      )
    );
  }

  try {
    await jwtVerify(
      token,
      secret
    );

    return NextResponse.next();
  } catch {
    return NextResponse.redirect(
      new URL(
        "/login",
        request.url
      )
    );
  }
}

import { SignJWT, jwtVerify } from "jose";
import { cookies } from "next/headers";

const secretValue = process.env.JWT_SECRET;

if (!secretValue) {
    throw new Error("JWT_SECRET is not configured.");
}

const secret = new TextEncoder().encode(secretValue);

export async function createToken(user) {
    return await new SignJWT({
        userId: user.user_id,
        username: user.username,
        roleId: user.role_id,
        roleName: user.role_name,
        companyId: user.company_id,
    })
        .setProtectedHeader({ alg: "HS256" })
        .setIssuedAt()
        .setExpirationTime("1d")
        .sign(secret);
}

export async function verifyToken(token) {
    try {
        const { payload } = await jwtVerify(token, secret);
        return payload;
    } catch {
        return null;
    }
}

export async function getCurrentUser() {
    const cookieStore = await cookies();
    const token = cookieStore.get("erp_token")?.value;

    if (!token) {
        return null;
    }

    return await verifyToken(token);
}

import pool from "@/lib/db";

export async function hasPermission(
  roleId,
  permissionKey,
  action = "view"
) {
  const result = await pool.query(
    `
    SELECT
      CASE
        WHEN $2 = 'view' THEN rp.can_view
        WHEN $2 = 'create' THEN rp.can_create
        WHEN $2 = 'update' THEN rp.can_update
        WHEN $2 = 'delete' THEN rp.can_delete
        ELSE FALSE
      END AS allowed

    FROM role_permissions rp

    INNER JOIN permissions p
      ON p.permission_id = rp.permission_id

    WHERE rp.role_id = $1
      AND p.permission_key = $3
    `,
    [
      roleId,
      action,
      permissionKey,
    ]
  );

  return result.rows[0]?.allowed === true;
}


export async function requirePermission(
  user,
  permissionKey,
  action = "view"
) {
  if (!user) {
    throw new Error("UNAUTHORIZED");
  }

  const allowed = await hasPermission(
    user.roleId,
    permissionKey,
    action
  );

  if (!allowed) {
    throw new Error("FORBIDDEN");
  }

  return true;
}

    useState(initialCompanyId);
  const [companies, setCompanies] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  async function handleLogin(event) {
    event.preventDefault();
    setError("");

    if (!username.trim() || !password) {
      setError("Please enter your username and password.");
      return;
    }

    setLoading(true);

    try {
      const response = await fetch("/api/auth/login", {
        method: "POST",
        credentials: "same-origin",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          username: username.trim(),
          password,
          companyId: selectedCompanyId || undefined,
        }),
      });

      const data = await response.json().catch(() => ({}));

      if (response.status === 409 && data.companies) {
        setCompanies(data.companies);
        setSelectedCompanyId("");
        setError("Your account has access to multiple companies. Select one to continue.");
        return;
      }

      if (!response.ok) {
        setError(data.message || "Login failed. Please check your credentials.");
        return;
      }

      const companyId = String(data.user.companyId);
      const companyName = data.user.companyName;

        "erp-selected-company",
        JSON.stringify({ companyId, companyName })

    const text = await response.text();

                                                            (value, index, array) =>

                                                                index

                                            </p>

                                        <Link

                    </div>

            </div>

}

            try {
                const response = await fetch("/api/companies/register", {
                    method: "GET",
                    cache: "no-store",
                    headers: { Accept: "application/json" },
                    signal: controller.signal,
                });

                const data = await readResponse(response);

                if (controller.signal.aborted) return;

                const availableUsers = Array.isArray(data.users)
                    ? data.users
                    : [];

                const availableRoles = Array.isArray(data.roles)
                    ? data.roles
                    : [];

                setUsers(availableUsers);
                setRoles(availableRoles);

                setForm((current) => {
                    const currentRoleExists = availableRoles.some(
                        (role) =>
                            String(role.role_id) === current.existingRoleId
                    );

                    const defaultRole =
                        availableRoles.find(
                            (role) =>
                                role.role_name?.toUpperCase() === "USER"
                        ) || availableRoles[0];

                    return {
                        ...current,
                        existingRoleId: currentRoleExists
                            ? current.existingRoleId
                            : String(defaultRole?.role_id ?? ""),
                    };
                });
            } catch (err) {
                if (
                    err.name !== "AbortError" &&
                    !controller.signal.aborted
                ) {
                    setError(
                        err.message ||
                            "Unable to load existing users and roles."
                    );
                }
            } finally {
                if (!controller.signal.aborted) {
                    setLoadingOptions(false);
                }
            }
        }

        Promise.resolve().then(() => {
            if (!controller.signal.aborted) {
                fetchOptions();
            }
        });

        return () => controller.abort();
    }, []);

        if (!file) return;

        if (!allowedLogoTypes.includes(file.type)) {
            setError("Choose a PNG, JPG, or WebP image.");
            event.target.value = "";
            return;
        }

        if (file.size > MAX_LOGO_SIZE) {
            setError("Company logo must be 2 MB or smaller.");
            event.target.value = "";
            return;
        }

        reader.onload = () => {
            if (typeof reader.result === "string") {
                setLogoFile(file);
                setLogoPreview(reader.result);
                setError("");
                setSuccess("");
            }
        };

        reader.onerror = () => {
            setError("Unable to read this image. Please try another file.");
            event.target.value = "";
        };

        reader.readAsDataURL(file);
    }

        setLogoFile(null);
        setLogoPreview("");

        setSuccess("");

        if (!form.companyName.trim()) {
            return "Company name is required.";
        }

        if (form.assignmentMode === "existingUser") {
            if (!form.existingUserId || !form.existingRoleId) {
                return "Select an existing user and their role.";
            }

            if (
                !form.companyPassword ||
                !form.confirmCompanyPassword
            ) {
                return "Enter and confirm the password for this company.";
            }

            if (form.companyPassword.length < 8) {
                return "Company password must be at least 8 characters long.";
            }

            if (form.companyPassword !== form.confirmCompanyPassword) {
                return "Company passwords do not match.";
            }
        } else {
            if (
                !form.fullName.trim() ||
                !form.username.trim() ||
                !form.adminEmail.trim() ||
                !form.password ||
                !form.confirmPassword
            ) {
                return "Complete all required administrator fields.";
            }

            if (form.password.length < 8) {
                return "Password must be at least 8 characters long.";
            }

            if (form.password !== form.confirmPassword) {
                return "Passwords do not match.";
            }
        }

        return "";
    }

async function readResponse(response) {
    const text = await response.text();
    let data = {};

    if (text) {
        try {
            data = JSON.parse(text);
        } catch {
            throw new Error(
                `Invalid server response (HTTP ${response.status}). Check the API route.`
            );
        }
    }

    if (!response.ok) {
        throw new Error(
            data.message || `Request failed (HTTP ${response.status}).`
        );
    }

    return data;
}

async function readResponse(response) {
    const text = await response.text();
    let data = {};

    if (text) {
        try {
            data = JSON.parse(text);
        } catch {
            throw new Error(
                `Invalid server response (HTTP ${response.status}). Check the API route.`
            );
        }
    }

    if (!response.ok) {
        throw new Error(
            data.message || `Request failed (HTTP ${response.status}).`
        );
    }

    return data;
}

export async function POST(request) {
  try {
    const body = await request.json();

    const username =
      typeof body.username === "string" ? body.username.trim() : "";
    const password = body.password;
    const companyId =
      body.companyId === undefined ||
      body.companyId === null ||
      body.companyId === ""
        ? null
        : Number(body.companyId);

    if (
      !username ||
      typeof password !== "string" ||
      !password ||
      (companyId !== null &&
        (!Number.isSafeInteger(companyId) || companyId <= 0))
    ) {
      return NextResponse.json(
        {
          success: false,
          message: "Enter a valid username and password.",
        },
        { status: 400 }
      );
    }

    const result = await pool.query(
      `
      SELECT
        u.user_id,
        u.full_name,
        u.username,
        u.email,
        u.is_active AS user_is_active,
        uc.company_id,
        uc.role_id,
        uc.company_password_hash,
        c.company_name,
        c.is_active AS company_is_active,
        r.role_name
      FROM users u
      JOIN user_companies uc
        ON uc.user_id = u.user_id
      JOIN companies c
        ON c.company_id = uc.company_id
      JOIN roles r
        ON r.role_id = uc.role_id
      WHERE LOWER(u.username) = LOWER($1)
        AND ($2::integer IS NULL OR uc.company_id = $2)
        AND (
          r.company_id IS NULL
          OR r.company_id = uc.company_id
        )
      ORDER BY uc.company_id
      `,
      [username, companyId]
    );

    const matchingUsers = [];

    for (const user of result.rows) {
      if (
        !user.user_is_active ||
        !user.company_is_active ||
        !user.company_password_hash
      ) {
        continue;
      }

      const passwordMatches = await bcrypt.compare(
        password,
        user.company_password_hash
      );

      if (passwordMatches) {
        matchingUsers.push(user);
      }
    }

    if (matchingUsers.length === 0) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid username, password, or company access.",
        },
        { status: 401 }
      );
    }

    if (companyId === null && matchingUsers.length > 1) {
      return NextResponse.json(
        {
          success: false,
          code: "COMPANY_SELECTION_REQUIRED",
          message: "Select the company you want to access.",
          companies: matchingUsers.map((user) => ({
            company_id: user.company_id,
            company_name: user.company_name,
          })),
        },
        { status: 409 }
      );
    }

    const user = matchingUsers[0];

    await pool.query(
      `
      UPDATE users
      SET last_login = CURRENT_TIMESTAMP,
          updated_at = CURRENT_TIMESTAMP
      WHERE user_id = $1
      `,
      [user.user_id]
    );

    // const token = await createToken({
    //   user_id: user.user_id,
    //   username: user.username,
    //   role_id: user.role_id,
    //   role_name: user.role_name,
    //   company_id: user.company_id,
    // });

    const token = await createToken({
  userId: user.user_id,
  username: user.username,
  roleId: user.role_id,
  roleName: user.role_name,
  companyId: user.company_id,
});

    const response = NextResponse.json({
      success: true,
      message: "Login successful.",
      user: {
        userId: user.user_id,
        fullName: user.full_name,
        username: user.username,
        roleName: user.role_name,
        companyId: user.company_id,
        companyName: user.company_name,
      },
    });

    response.cookies.set("erp_token", token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      maxAge: 60 * 60 * 24,
      path: "/",
    });

    return response;
  } catch (error) {
    console.error("Login error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "Server error during login.",
      },
      { status: 500 }
    );
  }
}

function validEmail(value) {

function parsePositiveId(value) {
    const id = Number(value);

async function saveCompanyLogo(file) {
    if (!file || typeof file.arrayBuffer !== "function" || file.size === 0) {
        return null;
    }

    const allowedTypes = {
        "image/png": ".png",
        "image/jpeg": ".jpg",
        "image/webp": ".webp",
    };

    if (!allowedTypes[file.type]) {
        const error = new Error("Choose a PNG, JPG, or WebP company logo.");
        error.status = 400;
        throw error;
    }

    if (file.size > 2 * 1024 * 1024) {
        const error = new Error("Company logo must be 2 MB or smaller.");
        error.status = 400;
        throw error;
    }

    const uploadDirectory = path.join(
        process.cwd(),
        "public",
        "uploads",
        "companies"
    );

    await mkdir(uploadDirectory, { recursive: true });

    const fileName = `${randomUUID()}${allowedTypes[file.type]}`;
    const filePath = path.join(uploadDirectory, fileName);

    try {
        await writeFile(
            filePath,
            Buffer.from(await file.arrayBuffer()),
            { flag: "wx" }
        );

        return {
            url: `/uploads/companies/${fileName}`,
            filePath,
        };
    } catch (error) {
        await unlink(filePath).catch(() => {});
        throw error;
    }
}

async function getCreatorAccess(userId) {
    const result = await pool.query(
        `
        SELECT uc.company_id, uc.role_id, r.role_name
        FROM user_companies uc
        INNER JOIN companies c ON c.company_id = uc.company_id
        INNER JOIN roles r ON r.role_id = uc.role_id
        WHERE uc.user_id = $1 AND c.is_active = TRUE
        ORDER BY uc.company_id;
        `,
        [userId]
    );

    const memberships = result.rows;

    if (!memberships.length) {
        return {
            memberships: [],
            allowed: false,
            isAdmin: false,
            creatorRoleName: null,
        };
    }

    let allowedMembership = null;
    let isAdmin = false;

    for (const membership of memberships) {
        const roleName = String(membership.role_name || "").toUpperCase();

        if (roleName === "ADMIN") {
            allowedMembership = membership;
            isAdmin = true;
            break;
        }

        const permitted = await hasPermission(
            membership.role_id,
            "companies",
            "create"
        );

        if (permitted && !allowedMembership) {
            allowedMembership = membership;
        }
    }

    return {
        memberships,
        allowed: Boolean(allowedMembership),
        isAdmin,
        creatorRoleName: allowedMembership?.role_name || null,
    };
}

async function getGlobalSystemRoles() {
    const result = await pool.query(

                WHEN 'USER' THEN 1
                WHEN 'SUB_ADMIN' THEN 2
                WHEN 'ADMIN' THEN 3
                ELSE 4
            END,
            role_name;
        `
    );

    return result.rows;
}

export async function POST(request) {
  try {
    // 1. Verify the current session
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        { message: "Unauthorized. Please log in again." },
        { status: 401 }
      );
    }

    // 2. Check create-role permission
    const allowed = await hasPermission(
      user.roleId,
      "roles",
      "create"
    );

    if (!allowed) {
      return NextResponse.json(
        {
          message: "You do not have permission to create roles.",
        },
        { status: 403 }
      );
    }

    // 3. Read request body
    const body = await request.json();

    const {
      roleName,
      description,
      authPassword,
    } = body;

    // 4. Validate role name
    if (
      typeof roleName !== "string" ||
      !roleName.trim()
    ) {
      return NextResponse.json(
        { message: "Role name is required." },
        { status: 400 }
      );
    }

    // 5. Require the current account password
    if (
      typeof authPassword !== "string" ||
      !authPassword
    ) {
      return NextResponse.json(
        {
          message: "Your current account password is required.",
        },
        { status: 400 }
      );
    }

    // 6. Verify the authenticated user's password
    // IMPORTANT: This assumes getCurrentUser() returns user.userId.
    if (!user.userId) {
      console.error(
        "getCurrentUser() does not provide userId."
      );

      return NextResponse.json(
        {
          message:
            "Unable to identify your account. Check lib/auth.js.",
        },
        { status: 500 }
      );
    }

    const passwordResult = await pool.query(
      `
      SELECT password_hash
      FROM users
      WHERE user_id = $1
      LIMIT 1;
      `,
      [user.userId]
    );

    if (passwordResult.rows.length === 0) {
      return NextResponse.json(
        { message: "Could not verify your account." },
        { status: 401 }
      );
    }

    const passwordHash =
      passwordResult.rows[0].password_hash;

    if (!passwordHash) {
      return NextResponse.json(
        {
          message: "Your account does not have a valid password hash.",
        },
        { status: 401 }
      );
    }

    const passwordMatches = await bcrypt.compare(
      authPassword,
      passwordHash
    );

    if (!passwordMatches) {
      return NextResponse.json(
        { message: "Incorrect account password." },
        { status: 401 }
      );
    }

    // 7. Create the role only after password verification
    const result = await pool.query(
      `
      INSERT INTO roles (
        role_name,
        description,
        is_system_role
      )
      VALUES ($1, $2, FALSE)
      RETURNING
        role_id,
        role_name,
        description,
        is_system_role,
        created_at;
      `,
      [
        roleName.trim().toUpperCase(),
        typeof description === "string" && description.trim()
          ? description.trim()
          : null,
      ]
    );

    return NextResponse.json(
      {
        success: true,
        message: "Role created successfully.",
        role: result.rows[0],
      },
      { status: 201 }
    );
  } catch (error) {
    console.error("CREATE ROLE ERROR:", error);

    // PostgreSQL unique constraint violation
    if (error.code === "23505") {
      return NextResponse.json(
        { message: "Role name already exists." },
        { status: 409 }
      );
    }

    return NextResponse.json(
      { message: "Failed to create role." },
      { status: 500 }
    );
  }
}
